# Vulkan-path golden traces of the shared video renderer
# (video_renderer_golden_vk.cc, harness in renderer_harness.h).
#
# One [scenario] per section, one expected line per line; lines starting
# with '#' are comments. Only ever read by the test: an expectation changes
# only by editing this file in a reviewed commit.
#
# Serials number the images FakeVulkanInterop created in this scenario, in
# CreateImage order from 1. `<device>` stands for the runner's Vulkan device
# name. `acquire index=-1 ... serial=none` is an AcquireFront with no front
# frame; index=-2 is the previous generation's retired front frame.

# A frame becomes the front only once its timeline value was signalled:
# while frame 1's GPU work is outstanding AcquireFront has nothing to show,
# after the signal it returns frame 1. Then three frames that complete at
# once rotate through the ring (latency 2, ring 6), and every image is
# destroyed exactly once at teardown.

# ADR 0004 reasons on the Vulkan path: one trail per candidate ranked
# before the winner, in chain order; get_info failing names the renderer
# "unknown renderer"; an explicit gpu-next leaves only its own trail.
[vk-reasons]
case gpu-next-create-fails
start ok backend=vulkan reason="vulkan (gpu) on <device>; vulkan/gpu-next unavailable: mpv_render_context_create: fake error -18"
create-attempts vulkan/gpu-next,vulkan/gpu
teardown frees=1 live-images=0
case both-creates-fail
start ok backend=software reason="software (sw) on cpu; vulkan/gpu-next unavailable: mpv_render_context_create: fake error -18; vulkan/gpu unavailable: mpv_render_context_create: fake error -15"
create-attempts vulkan/gpu-next,vulkan/gpu,sw
teardown frees=1 live-images=0
case device-not-shareable
start ok backend=software reason="software (sw) on cpu; vulkan/gpu-next unavailable: <device> is not the compositor adapter; vulkan/gpu unavailable: <device> is not the compositor adapter"
create-attempts sw
teardown frees=1 live-images=0
case get-info-fails
start ok backend=vulkan reason="vulkan (unknown renderer) on <device>"
create-attempts vulkan/gpu-next
teardown frees=1 live-images=0
case gpu-next-only-fails
start ok backend=software reason="software (sw) on cpu; vulkan/gpu-next unavailable: mpv_render_context_create: fake error -18"
create-attempts vulkan/gpu-next,sw
teardown frees=1 live-images=0
